cmake_minimum_required(VERSION 3.12)
project(GuardSight LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB_RECURSE GUARDSIGHT_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(guardsight )
target_include_directories(guardsight PUBLIC /include)
if(MSVC)
  target_compile_options(guardsight PRIVATE /W4)
else()
  target_compile_options(guardsight PRIVATE -Wall -Wextra -Wpedantic)
endif()
foreach(tool packetscan iocmatch rulecheck policyaudit)
  add_executable( tools/.cpp)
  target_link_libraries( PRIVATE guardsight)
endforeach()
enable_testing()
foreach(test core_tests packet_tests ioc_tests rules_tests policy_tests)
  if(test STREQUAL "core_tests")
    set(path tests/core/core_tests.cpp)
  elseif(test STREQUAL "packet_tests")
    set(path tests/packet/packet_tests.cpp)
  elseif(test STREQUAL "ioc_tests")
    set(path tests/ioc/ioc_tests.cpp)
  elseif(test STREQUAL "rules_tests")
    set(path tests/rules/rules_tests.cpp)
  else()
    set(path tests/policy/policy_tests.cpp)
  endif()
  add_executable( )
  target_link_libraries( PRIVATE guardsight)
  add_test(NAME  COMMAND )
endforeach()
